Arduino + Ultrasonic + BLE Sketch

Compatible with HC-08 / HM-10 / ESP32 / Nano 33 BLE

Supported BLE Serial Formats (auto-detected by this app):

  • D:14 or raw number 14 → Live ultrasonic distance in cm (smoothly moves man up/down & counts reps)
  • DOWN / UP → Directly animates stickman down or up
  • COUNT:5 or REP → Syncs exact rep count from Arduino
  • App sends RESET\n to Arduino when you press Reset.
// PushUp Sense - Arduino + HC-SR04 + BLE Module (HM-10 / HC-08)
// Wiring: Trig -> Pin 9, Echo -> Pin 10, BLE TX -> Pin 2, BLE RX -> Pin 3
#include <SoftwareSerial.h>

SoftwareSerial bleSerial(2, 3); // RX, TX
const int trigPin = 9;
const int echoPin = 10;

void setup() {
  Serial.begin(9600);
  bleSerial.begin(9600);
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
}

void loop() {
  // Check if Web App sent RESET command
  if (bleSerial.available()) {
    String cmd = bleSerial.readStringUntil('\n');
    cmd.trim();
    if (cmd == "RESET") {
      Serial.println("Counter reset by Web App");
    }
  }

  // Trigger HC-SR04 Ultrasonic Sensor
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH, 25000);
  int distanceCm = duration * 0.034 / 2;

  if (distanceCm > 2 && distanceCm < 100) {
    // Send distance over BLE to PushUp Sense Web App
    bleSerial.print("D:");
    bleSerial.println(distanceCm);
  }

  delay(80); // ~12 Hz smooth animation update rate
}
